# From Open-Meteo

`conditionFromCode` reads a WMO weather code; zip the arrays into objects and keep the units in °C and km/h.

```ts
import { conditionFromCode, type WeatherHour } from '@justanarthur/modo-ui-notch'

const res = await fetch(
  `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&timeformat=unixtime&timezone=auto` +
    '&current=temperature_2m,apparent_temperature,is_day,weather_code,wind_speed_10m,relative_humidity_2m,uv_index' +
    '&hourly=temperature_2m,weather_code,is_day,precipitation_probability' +
    '&daily=weather_code,temperature_2m_max,temperature_2m_min,precipitation_probability_max&forecast_days=2',
)
const { current, hourly, daily } = await res.json()

const hours: WeatherHour[] = hourly.time.map((time: number, i: number) => ({
  time: time * 1000,
  temperature: hourly.temperature_2m[i],
  condition: conditionFromCode(hourly.weather_code[i]),
  isDay: hourly.is_day[i] === 1,
  precipitation: hourly.precipitation_probability[i],
}))
```
