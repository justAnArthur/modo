import Tabs from './index'
import Badge from '../../badge'
import { Clock, Settings, SquareLibrary, Star } from 'lucide-react'

# Basic

The indicator slides to the clicked tab; the hover pill previews the next one.

<div className="flex flex-col gap-4 w-full">
  <Tabs defaultValue="library">
    <Tabs.List>
      <Tabs.Item value="library" label="Library" />
      <Tabs.Item value="recents" label="Recents" />
      <Tabs.Item value="favorites" label="Favorites" />
      <Tabs.Item value="settings" label="Settings" />
    </Tabs.List>
    <Tabs.Panel value="library"><p className="text-body text-muted-foreground pt-3">Library content.</p></Tabs.Panel>
    <Tabs.Panel value="recents"><p className="text-body text-muted-foreground pt-3">Recents content.</p></Tabs.Panel>
    <Tabs.Panel value="favorites"><p className="text-body text-muted-foreground pt-3">Favorites content.</p></Tabs.Panel>
    <Tabs.Panel value="settings"><p className="text-body text-muted-foreground pt-3">Settings content.</p></Tabs.Panel>
  </Tabs>
</div>

# With Icons

A leading `icon` thickens its stroke while its tab is hovered or active.

<div className="flex flex-col gap-4 w-full">

# Uncontrolled

FF's Controlled section, without the state: `defaultValue` picks the first
active tab and the tabs keep their own state. To control them, pass `value`
+ `onValueChange` (or `selectedIndex` + `onSelect`) instead.

<div className="flex flex-col gap-4 w-full">
  <Tabs defaultValue="recents">
    <Tabs.List>
      <Tabs.Item value="library" label="Library" />
      <Tabs.Item value="recents" label="Recents" />
      <Tabs.Item value="favorites" label="Favorites" />
    </Tabs.List>
    <Tabs.Panel value="library"><p className="text-body text-muted-foreground pt-3">Library content.</p></Tabs.Panel>
    <Tabs.Panel value="recents"><p className="text-body text-muted-foreground pt-3">Recents content.</p></Tabs.Panel>
    <Tabs.Panel value="favorites"><p className="text-body text-muted-foreground pt-3">Favorites content.</p></Tabs.Panel>
  </Tabs>
</div>

# With counters

A `label` can be any node: text with a counter or a Badge. It still holds its width as the tab turns semibold.

<div className="flex flex-col gap-4 w-full">
  <Tabs defaultValue="open">
    <Tabs.List>
      <Tabs.Item value="open" label={<>Open <Badge size="compact">12</Badge></>} />
      <Tabs.Item value="review" label={<>In review <Badge size="compact">3</Badge></>} />
      <Tabs.Item value="closed" label={<>Closed <span className="tabular-nums text-muted-foreground">148</span></>} />
    </Tabs.List>
    <Tabs.Panel value="open"><p className="text-body text-muted-foreground pt-3">Twelve open requests.</p></Tabs.Panel>
    <Tabs.Panel value="review"><p className="text-body text-muted-foreground pt-3">Three waiting for review.</p></Tabs.Panel>
    <Tabs.Panel value="closed"><p className="text-body text-muted-foreground pt-3">Closed requests.</p></Tabs.Panel>
  </Tabs>
</div>

# Link tabs

`href` renders a tab as an anchor, for tabs that are pages: set the root's `value` from the current route and the indicator sits on it.

<Tabs defaultValue="overview">
  <Tabs.List>
    <Tabs.Item value="overview" label="Overview" href="#overview" />
    <Tabs.Item value="activity" label="Activity" href="#activity" />
    <Tabs.Item value="settings" label="Settings" href="#settings" />
  </Tabs.List>
</Tabs>
