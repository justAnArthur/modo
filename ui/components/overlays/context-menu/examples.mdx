import ContextMenu from './index'
import { ArrowLeft, ArrowRight, Code, Copy, Link, Printer, RotateCw, Save, Scissors, Clipboard, Trash2, Share, Mail, Image, Users } from 'lucide-react'

# Right click

Right-click anywhere in the area (long-press on touch): the menu swells out of the exact point you pressed. Rows are Dropdown's, with a shortcut at the end.

<ContextMenu>
  <ContextMenu.Trigger className="flex flex-wrap h-56 w-full max-w-lg items-center justify-center rounded-xl border border-dashed border-border text-caption text-muted-foreground">
    Right-click here
  </ContextMenu.Trigger>
  <ContextMenu.Content>
    <ContextMenu.Item index={0} icon={ArrowLeft} label="Back" shortcut="⌘[" />
    <ContextMenu.Item index={1} icon={ArrowRight} label="Forward" shortcut="⌘]" disabled />
    <ContextMenu.Item index={2} icon={RotateCw} label="Reload" shortcut="⌘R" />
    <ContextMenu.Separator />
    <ContextMenu.Item index={3} icon={Save} label="Save as…" shortcut="⌘S" />
    <ContextMenu.Item index={4} icon={Printer} label="Print…" shortcut="⌘P" />
    <ContextMenu.Item index={5} icon={Link} label="Copy link" />
    <ContextMenu.Separator />
    <ContextMenu.Label>Developer</ContextMenu.Label>
    <ContextMenu.Item index={6} icon={Code} label="Inspect" shortcut="⌥⌘I" />
  </ContextMenu.Content>
</ContextMenu>

# Submenus

`ContextMenu.Sub` nests a menu, as Dropdown's does: its `ContextMenu.SubTrigger` row opens a `ContextMenu.SubContent` beside it when the pointer rests on it, on a press or on ArrowRight, grown out of that row; ArrowLeft and Escape close one level. Submenus nest, and each runs its own selection; `keepMounted` keeps it across opens.

<ContextMenu>
  <ContextMenu.Trigger className="flex flex-wrap h-56 w-full max-w-lg items-center justify-center rounded-xl border border-dashed border-border text-caption text-muted-foreground">
    Right-click to share
  </ContextMenu.Trigger>
  <ContextMenu.Content keepMounted>
    <ContextMenu.Item icon={Copy} label="Copy" shortcut="⌘C" />
    <ContextMenu.Item icon={Scissors} label="Cut" shortcut="⌘X" />
    <ContextMenu.Separator />
    <ContextMenu.Sub>
      <ContextMenu.SubTrigger icon={Share} label="Share" />
      <ContextMenu.SubContent>
        <ContextMenu.Item icon={Link} label="Copy link" />
        <ContextMenu.Item icon={Mail} label="Email" />
        <ContextMenu.Sub>
          <ContextMenu.SubTrigger icon={Image} label="Export as" />
          <ContextMenu.SubContent defaultCheckedIndex={0}>
            <ContextMenu.Item label="PDF" />
            <ContextMenu.Item label="PNG" />
            <ContextMenu.Item label="Markdown" />
          </ContextMenu.SubContent>
        </ContextMenu.Sub>
      </ContextMenu.SubContent>
    </ContextMenu.Sub>
    <ContextMenu.Sub disabled>
      <ContextMenu.SubTrigger icon={Users} label="Move to team" disabled />
      <ContextMenu.SubContent>
        <ContextMenu.Item label="Design" />
      </ContextMenu.SubContent>
    </ContextMenu.Sub>
    <ContextMenu.Separator />
    <ContextMenu.Item icon={Trash2} label="Delete" shortcut="⌫" />
  </ContextMenu.Content>
</ContextMenu>

# Compact

`size="compact"` pins the rows to the 28px step of the size ladder, for dense surfaces.

<ContextMenu size="compact">
  <ContextMenu.Trigger className="flex flex-wrap h-40 w-full max-w-lg items-center justify-center rounded-xl border border-dashed border-border text-caption text-muted-foreground">
    Right-click a file
  </ContextMenu.Trigger>
  <ContextMenu.Content className="w-48">
    <ContextMenu.Item index={0} icon={Scissors} label="Cut" shortcut="⌘X" />
    <ContextMenu.Item index={1} icon={Copy} label="Copy" shortcut="⌘C" />
    <ContextMenu.Item index={2} icon={Clipboard} label="Paste" shortcut="⌘V" disabled />
    <ContextMenu.Separator />
    <ContextMenu.Item index={3} icon={Trash2} label="Delete" shortcut="⌫" />
  </ContextMenu.Content>
</ContextMenu>

# Morph

`effect="morph"` grows the menu out of the point as one clean surface, without the goo.

<ContextMenu>
  <ContextMenu.Trigger className="flex flex-wrap h-40 w-full max-w-lg items-center justify-center rounded-xl border border-dashed border-border text-caption text-muted-foreground">
    Right-click for a plain morph
  </ContextMenu.Trigger>
  <ContextMenu.Content effect="morph">
    <ContextMenu.Item index={0} icon={Copy} label="Copy" shortcut="⌘C" />
    <ContextMenu.Item index={1} icon={Link} label="Copy link" />
    <ContextMenu.Item index={2} icon={Printer} label="Print…" shortcut="⌘P" />
  </ContextMenu.Content>
</ContextMenu>

# Selection

Dropdown's selection props work here too. `defaultCheckedIndex` makes the rows radio options (`menuitemradio`): the menu moves the check itself.

<ContextMenu>
  <ContextMenu.Trigger className="flex flex-wrap h-40 w-full max-w-lg items-center justify-center rounded-xl border border-dashed border-border text-caption text-muted-foreground">
    Right-click to sort
  </ContextMenu.Trigger>
  <ContextMenu.Content defaultCheckedIndex={0}>
    <ContextMenu.Label>Sort by</ContextMenu.Label>
    <ContextMenu.Item index={0} label="Name" />
    <ContextMenu.Item index={1} label="Date modified" />
    <ContextMenu.Item index={2} label="Size" />
  </ContextMenu.Content>
</ContextMenu>
