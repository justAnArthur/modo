import DatePicker from './index'
import { InputGroup } from '../../input'
import { ar, de } from 'react-day-picker/locale'

# Basic

Press the field: the date is selected to type over, and the calendar scrolls beneath it.

<div className="w-72 max-w-full">
  <DatePicker placeholder="Pick a date" />
</div>

# With time

`withTime` adds the time to the field and hour and minute columns to the panel.

<div className="w-72 max-w-full">
  <DatePicker withTime defaultValue={new Date(2026, 9, 9, 15, 33)} />
</div>

# Editable

`editable` makes the field a text input: type the digits and the separators fall into place, the rest of the mask muted after the caret. A date that can't exist (31.02), or a weekend here, marks the field invalid; Escape brings back the last real date. The calendar icon opens the panel.

<div className="w-72 max-w-full">
  <DatePicker editable disabledDays={{ dayOfWeek: [0, 6] }} defaultValue={new Date(2026, 9, 9)} />
</div>

# Editable, German, with time

The mask follows the locale's order and separators ("DD.MM.YYYY, HH:mm"); a time is typed on a 24-hour clock.

<div className="w-72 max-w-full">
  <DatePicker editable withTime locale={de} defaultValue={new Date(2026, 9, 9, 15, 33)} />
</div>

# String values

`valueType="string"` takes and reports ISO strings, the format `name` submits: `"2026-10-09"`, or `"2026-10-09T15:33"` with `withTime`, read as local time. A string that reads as no date marks the field invalid.

<div className="flex w-72 max-w-full flex-col gap-2">
  <DatePicker valueType="string" defaultValue="2026-10-09" />
  <DatePicker valueType="string" withTime editable defaultValue="2026-10-09T15:33" />
  <DatePicker valueType="string" defaultValue="not a date" placeholder="Unreadable value" />
</div>

```tsx
const [due, setDue] = useState<string | null>('2026-10-09')

<DatePicker valueType="string" value={due} onValueChange={setDue} />
```

# Your own text

`formatDate` writes the field's text; `labels` translates every word, the caption under a date that can't exist (`labels.invalid`) included.

<div className="flex w-72 max-w-full flex-col gap-2">
  <DatePicker
    formatDate={date => date.toLocaleDateString('en-GB', { weekday: 'long', day: 'numeric', month: 'long' })}
    defaultValue={new Date(2026, 9, 9)}
  />
  <DatePicker editable labels={{ date: 'Datum', calendar: 'Kalender öffnen', invalid: 'Kein gültiges Datum' }} locale="de" />
</div>

# Gaps, required, unselectable

`hidden` days leave gaps and can't be typed; `required` and `unselectable` keep the picked day when it is pressed again.

<div className="w-72 max-w-full">
  <DatePicker hidden={{ dayOfWeek: [0, 6] }} unselectable defaultValue={new Date(2026, 9, 9)} />
</div>

# Form wiring

`id`, the `aria-*` props and `onBlur` reach the field's control (its button, or an `editable` field's input), and `controlProps` adds a `ref` and any other attribute. `required` stops an empty native submit.

```tsx
<DatePicker
  id={id}
  name="due"
  required
  aria-describedby={hintId}
  aria-invalid={!!error}
  onBlur={field.onBlur}
  controlProps={{ ref: field.ref, 'data-testid': 'due' }}
/>
```

# German

<div className="w-72 max-w-full">
  <DatePicker locale={de} withTime defaultValue={new Date(2026, 9, 9, 15, 33)} />

# 12-hour clock

`hourCycle="h12"` reads the hours with AM/PM; `minuteStep` thins the minutes.

<div className="w-72 max-w-full">
  <DatePicker withTime hourCycle="h12" minuteStep={5} defaultValue={new Date(2026, 9, 9, 15, 35)} />
</div>

# Columns on both sides

`columnsSide="start"` puts the month and year before the days; the time stays after them.

<div className="w-72 max-w-full">
  <DatePicker withTime columnsSide="start" defaultValue={new Date(2026, 9, 9, 15, 33)} />
</div>

# In place

`effect="morph"` with `hideSource` turns the field itself into the panel.

<div className="w-72 max-w-full">
  <DatePicker effect="morph" hideSource defaultValue={new Date(2026, 9, 9)} />
</div>

# Right to left

`dir="rtl"` mirrors the panel for an Arabic locale.

<div className="w-72 max-w-full">
  <DatePicker locale={ar} dir="rtl" defaultValue={new Date(2026, 9, 9)} />
</div>

# Disabled days and bounds

`disabledDays` takes react-day-picker matchers; `startMonth` and `endMonth` bound the calendar and its year column.

<div className="w-72 max-w-full">
  <DatePicker
    placeholder="Weekdays in 2026"
    disabledDays={{ dayOfWeek: [0, 6] }}
    startMonth={new Date(2026, 0, 1)}
    endMonth={new Date(2026, 11, 1)}
  />
</div>

# A date of birth

`defaultMonth` is where the calendar opens while nothing is picked, here thirty years back; `endMonth` and `disabledDays` stop it at today.

<div className="w-72 max-w-full">
  <DatePicker
    placeholder="Date of birth"
    defaultMonth={new Date(new Date().getFullYear() - 30, 0, 1)}
    endMonth={new Date()}
    disabledDays={{ after: new Date() }}
  />
</div>

# Invalid and disabled

`invalid` turns the field destructive; `disabled` turns it off.

<div className="flex w-72 max-w-full flex-col gap-2">
  <DatePicker invalid defaultValue={new Date(2026, 9, 9)} />
  <DatePicker disabled defaultValue={new Date(2026, 9, 9)} />
</div>

# In a form field

Inside an `InputGroup.Field`, the field's `error` marks the picker invalid and `disabled` turns it off; `name` submits the date as `yyyy-MM-dd`.

<InputGroup>
  <InputGroup.Field label="Due date" error="Pick a weekday.">
    <DatePicker name="due" defaultValue={new Date(2026, 9, 10)} />
  </InputGroup.Field>
  <InputGroup.Field label="Start date" disabled>
    <DatePicker name="start" />
  </InputGroup.Field>
</InputGroup>

# Compact

`size="compact"` pins the field and its panel to the compact step.

<div className="w-60 max-w-full">
