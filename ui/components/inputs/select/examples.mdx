import Select from './index'
import { Bell, Globe, Mail, Monitor, Moon, Shield, Sun, User } from 'lucide-react'
import Badge from '../../data-display/badge'
import Button from '../../actions/button'

# Basic

A trigger, a popover, four options. Rows number themselves in document
order, so they need no `index`.

<Select defaultValue="banana">
  <Select.Trigger placeholder="Select a fruit…" />
  <Select.Content>
    <Select.Item value="apple">Apple</Select.Item>
    <Select.Item value="banana">Banana</Select.Item>
    <Select.Item value="cherry">Cherry</Select.Item>
    <Select.Item value="mango">Mango</Select.Item>
  </Select.Content>
</Select>

# Variants

Bordered is the default; borderless drops the resting border and keeps the
hover fill, for toolbars and dense headers.

<div className="flex flex-wrap items-center gap-3">
  <Select>
    <Select.Trigger variant="bordered" placeholder="Bordered" />
    <Select.Content>
      <Select.Item value="apple">Apple</Select.Item>
      <Select.Item value="banana">Banana</Select.Item>
      <Select.Item value="cherry">Cherry</Select.Item>
    </Select.Content>
  </Select>
  <Select>
    <Select.Trigger variant="borderless" placeholder="Borderless" />
    <Select.Content>
      <Select.Item value="apple">Apple</Select.Item>
      <Select.Item value="banana">Banana</Select.Item>
      <Select.Item value="cherry">Cherry</Select.Item>
    </Select.Content>
  </Select>
</div>

# With Icons

A leading icon on the trigger and on every row. Row icons thicken to a 2
stroke while their row is hovered or checked.

<Select defaultValue="system">
  <Select.Trigger icon={Monitor} placeholder="Theme" />
  <Select.Content>
    <Select.Item value="system" icon={Monitor}>System</Select.Item>
    <Select.Item value="light" icon={Sun}>Light</Select.Item>
    <Select.Item value="dark" icon={Moon}>Dark</Select.Item>
  </Select.Content>
</Select>

# Groups

Select.Group + Select.Label caption a run of rows, Select.Separator rules
between them. The rows keep counting across the groups.

<Select>
  <Select.Trigger placeholder="Settings…" />
  <Select.Content>
    <Select.Group>
      <Select.Label>Account</Select.Label>
      <Select.Item value="profile" icon={User}>Profile</Select.Item>
      <Select.Item value="email" icon={Mail}>Email</Select.Item>
    </Select.Group>
    <Select.Separator />
    <Select.Group>
      <Select.Label>Preferences</Select.Label>
      <Select.Item value="notifications" icon={Bell}>Notifications</Select.Item>
      <Select.Item value="privacy" icon={Shield}>Privacy</Select.Item>
    </Select.Group>
  </Select.Content>
</Select>

# Scrollable List

Past 300px the list scrolls inside a ScrollArea: hover-revealed thumb, a
fade at both edges, and the overlays scrolling with the rows.

  <Select.Trigger icon={Globe} placeholder="Select timezone…" />
  <Select.Content>
    {[
      ['utc-8', '(UTC-8) Pacific Time'],
      ['utc-7', '(UTC-7) Mountain Time'],

# Error State

`error` on the trigger tints its border, sets `aria-invalid`, and prints
the message underneath.

<Select>
  <Select.Trigger placeholder="Select a role…" error="Please select a role to continue." />
  <Select.Content>
    <Select.Item value="admin">Admin</Select.Item>
    <Select.Item value="editor">Editor</Select.Item>
    <Select.Item value="viewer">Viewer</Select.Item>
  </Select.Content>
</Select>

# Disabled

`disabled` on the root takes the whole control out; on a single item it
dims the row and makes it invisible to the hover highlight.

<div className="flex flex-wrap items-center gap-3">
  <Select disabled>
    <Select.Trigger placeholder="Disabled" />
    <Select.Content>
      <Select.Item value="a">Option A</Select.Item>
    </Select.Content>
  </Select>
  <Select>
    <Select.Trigger placeholder="Some disabled…" />
    <Select.Content>
      <Select.Item value="a">Available</Select.Item>
      <Select.Item value="b" disabled>Unavailable</Select.Item>
      <Select.Item value="c">Available</Select.Item>
    </Select.Content>
  </Select>
</div>

# Morph

The menu oozes out of its trigger by default. `effect="morph"` with `hideSource` turns the trigger itself into the list and shrinks it back, one continuous surface.

<div className="flex flex-wrap items-start justify-center gap-3">
  <Select defaultValue="goo">
    <Select.Trigger placeholder="Goo" />
    <Select.Content>
      <Select.Item value="goo">Goo</Select.Item>
      <Select.Item value="neck">Liquid neck</Select.Item>
      <Select.Item value="drop">Drop</Select.Item>
    </Select.Content>
  </Select>
  <Select defaultValue="morph">
    <Select.Trigger placeholder="Morph" />
    <Select.Content effect="morph" hideSource>
      <Select.Item value="morph">Morph</Select.Item>
      <Select.Item value="grow">Grow</Select.Item>
      <Select.Item value="shrink">Shrink back</Select.Item>
    </Select.Content>
  </Select>
</div>

# Node labels

A row's label may be any node: it keeps its own box in the row and in the
trigger, where text is trimmed and truncated. `textValue` is what typeahead
matches. `Select.Value` as the trigger's children takes its own placeholder.

<Select defaultValue="beta">
  <Select.Trigger className="w-[220px]">
    <Select.Value placeholder="Release channel…" />
  </Select.Trigger>
  <Select.Content>
    <Select.Item value="stable" textValue="Stable">
      <span className="flex items-center gap-2">Stable <Badge color="blue" size="compact">1.4</Badge></span>
    </Select.Item>
    <Select.Item value="beta" textValue="Beta">
      <span className="flex items-center gap-2">Beta <Badge color="blue" size="compact">1.5</Badge></span>
    </Select.Item>
    <Select.Item value="canary" textValue="Canary">
      <span className="flex items-center gap-2">Canary <Badge color="blue" size="compact">nightly</Badge></span>
    </Select.Item>
  </Select.Content>
</Select>

# Your own trigger

`render` or `asChild` make your own element the trigger, look included.
Childless it keeps the value and the chevron; with children, those are its
whole body.

<div className="flex flex-wrap items-center gap-3">
  <Select defaultValue="week">
    <Select.Trigger render={<Button variant="tertiary" />} />
    <Select.Content>
      <Select.Item value="day">Today</Select.Item>
      <Select.Item value="week">This week</Select.Item>
      <Select.Item value="month">This month</Select.Item>
    </Select.Content>
  </Select>
  <Select defaultValue="asc">
    <Select.Trigger asChild>
      <Button variant="ghost">Sort</Button>
    </Select.Trigger>
    <Select.Content>
      <Select.Item value="asc">Oldest first</Select.Item>
      <Select.Item value="desc">Newest first</Select.Item>
    </Select.Content>
  </Select>
</div>

# Read-only

`readOnly` shows the value and keeps the popup shut, for a form you may
look at but not change.

<Select readOnly defaultValue="editor">
  <Select.Trigger placeholder="Role" />
  <Select.Content>
    <Select.Item value="admin">Admin</Select.Item>
    <Select.Item value="editor">Editor</Select.Item>
    <Select.Item value="viewer">Viewer</Select.Item>
  </Select.Content>
</Select>
