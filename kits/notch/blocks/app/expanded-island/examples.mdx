# Every widget

# Three tabs

`tabs` takes each tab's icon, label and view.

<ScreenEdge height={260}>
  <Island
    defaultState="expanded"
    collapsedState="idle"
    expanded={
      <ExpandedIsland
        defaultValue="calendar"
        tabs={[
          { value: 'music', label: 'Now Playing', icon: Music, content: <NowPlaying track={PLAYING} /> },
          { value: 'calendar', label: 'Calendar', icon: CalendarDays, content: <Agenda events={EVENTS} now={AGENDA_NOW} /> },
          { value: 'weather', label: 'Weather', icon: CloudSun, content: <Weather {...FORECAST} /> },
        ]}
      />
    }
  />
</ScreenEdge>
