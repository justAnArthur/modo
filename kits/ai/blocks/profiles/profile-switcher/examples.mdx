import ProfileSwitcher from './index'
import { useState } from 'react'
import { PROFILES } from '../../../mock/mock'

export function WithActions() {
  const [last, setLast] = useState('Nothing pressed yet')
  return (
    <div className="flex h-[560px] w-full flex-col items-center gap-3 pt-6">
      <ProfileSwitcher
        profiles={PROFILES}
        onAddProfile={() => setLast('Add profile: open the add-profile flow')}
        onManage={() => setLast('Manage: open the profiles screen')}
      />
      <code className="font-mono text-caption text-muted-foreground">{last}</code>
    </div>
  )
}

# Switcher

Press the trigger: the panel grows out of it. Pick a profile and its background melts over; the arrow keys move the pick.

<div className="flex h-[560px] w-full items-start justify-center pt-6">
  <ProfileSwitcher profiles={PROFILES} />
</div>

# Add and manage

`onAddProfile` and `onManage` put "Add profile" and "Manage" in the panel's footer; each shows only when its handler is given. A press closes the panel, then calls the handler.

<WithActions />

# Near a limit

`work` starts active here: its session meter is past 90%, so it reads red before you start anything there.

<div className="flex h-[560px] w-full items-start justify-center pt-6">
  <ProfileSwitcher profiles={PROFILES} defaultValue="work" />
</div>

# No usage yet

A profile has no limits until its first turn reports usage (`limits: []`, or every limit at 0): the trigger shows no usage lines, its row a dash instead of a share, and Limits says so.

<div className="flex h-[560px] w-full items-start justify-center pt-6">
  <ProfileSwitcher
    profiles={[...PROFILES, { ...PROFILES[2], id: 'fresh', name: 'fresh', configDir: '~/.claude-fresh', limits: [] }]}
    defaultValue="fresh"
  />
</div>
