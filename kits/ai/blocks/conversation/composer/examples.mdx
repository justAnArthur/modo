import Composer from './index'
import Button from '@justanarthur/modo-ui/components/actions/button'
import { Plus } from 'lucide-react'
import { useState } from 'react'
import { COMPOSER, MANY_PROJECTS, ZCODE_MODELS } from '../../../mock/mock'

export function ManyProjects() {
  const [projects, setProjects] = useState(MANY_PROJECTS)
  return (
    <div className="flex h-[480px] w-full max-w-2xl items-end">
      <Composer
        {...COMPOSER}
        target
        context={0}
        projects={projects}
        onRemoveProject={path => setProjects(all => all.filter(p => p.path !== path))}
        targetOptions={{ worktree: false, baseBranch: false, setupScript: false }}
      />
    </div>
  )
}

export function ControlledSettings() {
  const [project, setProject] = useState(COMPOSER.projects[0].path)
  const [worktree, setWorktree] = useState(true)
  const [baseBranch, setBaseBranch] = useState(COMPOSER.branches[0])
  const [profile, setProfile] = useState(COMPOSER.profiles[0].id)
  const [model, setModel] = useState('Sonnet 5.5')
  const [effort, setEffort] = useState('High')
  const [permissionMode, setPermissionMode] = useState('Accept edits')
  const [fast, setFast] = useState(false)
  const [outputStyle, setOutputStyle] = useState('Default')
  return (
    <div className="flex h-[240px] w-full max-w-2xl flex-col justify-end gap-3">
      <Composer
        {...COMPOSER}
        target
        context={0}
        project={project}
        onProjectChange={setProject}
        worktree={worktree}
        onWorktreeChange={setWorktree}
        baseBranch={baseBranch}
        onBaseBranchChange={setBaseBranch}
        profile={profile}
        onProfileChange={setProfile}
        model={model}
        onModelChange={setModel}
        effort={effort}
        onEffortChange={setEffort}
        permissionMode={permissionMode}
        onPermissionModeChange={setPermissionMode}
        fast={fast}
        onFastChange={setFast}
        outputStyle={outputStyle}
        onOutputStyleChange={setOutputStyle}
      />
      <code className="font-mono text-caption text-muted-foreground">
        {profile} · {project}{worktree ? ` (worktree from ${baseBranch})` : ''} · --model "{model}" --effort "{effort}"
        --permission-mode "{permissionMode}"{fast ? ' · fast' : ''} · {outputStyle}
      </code>
    </div>
  )
}

# Composer

In the draft, the + palette and the run trigger (permission mode, model, effort, context ring) that opens the run settings. On the shell, where the session runs: the project and the profile with its usage lines.

<div className="flex h-[440px] w-full max-w-2xl items-end">
  <Composer {...COMPOSER} />
</div>

# New session

`target`: the project (its popover holds the worktree, setup script and base branch) and the profile can still be picked. With `onAddProfile` and `onManageProfiles`, the profile panel's footer shows "Add profile" and "Manage"; a press closes the panel, then calls them.

<div className="flex h-[200px] w-full max-w-2xl items-end">
  <Composer
    {...COMPOSER}
    target
    context={0}
    placeholder="Describe the task, or type / for commands"
    onAddProfile={() => {}}
    onManageProfiles={() => {}}
  />
</div>

# Controlled settings

Every choice on the row is controllable: the caller holds the project (by its path), the worktree, the base branch, the profile, the model, the effort, the permission mode, fast mode and the output style, and passes them on to the agent.

<ControlledSettings />

# Many projects

Past six projects the project popover takes a search field: type to filter by name or path, Enter picks the first match, ↓ moves into the list, whose arrow keys move the pick. The list scrolls, and a long path is clipped from its start, keeping the folders nearest the project. `onRemoveProject` adds a Remove from the list action to each row.

<ManyProjects />

# Queued messages

Typed while a reply streams, messages wait above the draft. Drag a row (or Alt+↑/↓) to reorder, double-click to pull it back into the draft, × to drop it. Stop the reply and the top one sends.

<div className="flex h-[280px] w-full max-w-2xl items-end">
  <Composer
    {...COMPOSER}
    defaultStatus="streaming"
    defaultQueue={[
      { id: 'q1', text: 'Also add a dark-mode screenshot', files: [] },
      { id: 'q2', text: 'And link the changelog entry', files: [] },
    ]}
  />
</div>

# History

With the caret on the first line, ↑ recalls the previous message; ↓ on the last line walks forward and, past the newest, brings back the draft.

An empty draft offers prompts: Tab fills the ghost one, ↓ / ↑ walk the list under the draft (the text keeps focus), Enter or a click fills the highlighted one. Typing folds the list away.

# Draft alone

`Composer.Draft` without the shell and its menus, for a reply box on a board or a card. Its slots take any controls; a render function gets `openFilePicker`. `defaultFiles` turns attachments on: drop a PNG, JPEG or PDF on it, hover a tile for its ×.

<div className="w-full max-w-xl">
  <Composer.Draft
    defaultFiles={[
      new File(['<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 80 80"><rect width="80" height="80" /></svg>'], 'swatch.svg', { type: 'image/svg+xml' }),
      new File(['Quarterly notes'], 'notes.txt', { type: 'text/plain' }),
    ]}
    leftSlot={({ openFilePicker }) => (
      <Button variant="ghost" size="icon-compact" aria-label="Attach" onClick={() => openFilePicker()}>
        <Plus />
      </Button>
    )}
  />
</div>
