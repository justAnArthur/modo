import DatePicker from './index'
import { Label } from '../../input'
import Button from '../../../button'
import Dialog from '../../../overlays/dialog'
import { ar, de } from 'react-day-picker/locale'

# Basic

Press the field: the date is selected to type over, and the calendar scrolls beneath it.

<div className="w-72 max-w-full">
  <DatePicker placeholder="Pick a date" />
</div>

# With time

`withTime` adds the time to the field and hour and minute columns to the panel.

<div className="w-72 max-w-full">
  <DatePicker withTime defaultValue={new Date(2026, 9, 9, 15, 33)} />
</div>

# German

The locale sets the names, the formats shown and read ("12.10.2026"), and the week's first day.

<div className="w-72 max-w-full">
  <DatePicker locale={de} withTime defaultValue={new Date(2026, 9, 9, 15, 33)} />
</div>

# 12-hour clock

`hourCycle="h12"` reads the hours with AM/PM; `minuteStep` thins the minutes.

<div className="w-72 max-w-full">
  <DatePicker withTime hourCycle="h12" minuteStep={5} defaultValue={new Date(2026, 9, 9, 15, 35)} />
</div>

# Columns on both sides

`columnsSide="start"` puts the month and year before the days; the time stays after them.

<div className="w-72 max-w-full">
  <DatePicker withTime columnsSide="start" defaultValue={new Date(2026, 9, 9, 15, 33)} />
</div>

# In place

`effect="morph"` with `hideSource` turns the field itself into the panel.

<div className="w-72 max-w-full">
  <DatePicker effect="morph" hideSource defaultValue={new Date(2026, 9, 9)} />
</div>

# Right to left

`dir="rtl"` mirrors the panel for an Arabic locale.

<div className="w-72 max-w-full">
  <DatePicker locale={ar} dir="rtl" defaultValue={new Date(2026, 9, 9)} />
</div>

# Disabled days and bounds

`disabled` takes react-day-picker matchers; `startMonth` and `endMonth` bound the calendar and its year column.

<div className="w-72 max-w-full">
  <DatePicker
    placeholder="Weekdays in 2026"
    disabled={{ dayOfWeek: [0, 6] }}
    startMonth={new Date(2026, 0, 1)}
    endMonth={new Date(2026, 11, 1)}
  />
</div>

# Invalid and disabled

`invalid` turns the field destructive; `disabled={true}` turns it off.

<div className="flex w-72 max-w-full flex-col gap-2">
  <DatePicker invalid defaultValue={new Date(2026, 9, 9)} />
  <DatePicker disabled defaultValue={new Date(2026, 9, 9)} />
</div>

# Compact

`size="compact"` pins the field and its panel to the compact step.

<div className="w-60 max-w-full">
  <DatePicker size="compact" defaultValue={new Date(2026, 9, 9)} />
</div>

# Editable

`editable` makes the field a text input: type the digits and the separators fall into place, with the rest of the mask muted after the caret. The calendar icon opens the panel. A day that can't exist, or one left half typed, turns the field destructive with a caption.

<div className="flex w-72 max-w-full flex-col gap-1">
  <Label htmlFor="editable-date">Date of birth</Label>
  <DatePicker id="editable-date" editable defaultValue={new Date(1990, 4, 17)} />
</div>

# Editable with time

With `withTime` the mask takes the time too, on a 24-hour clock; the locale sets the order ("17.05.1990, 14:30").

<div className="w-72 max-w-full">
  <DatePicker editable withTime locale={de} defaultValue={new Date(1990, 4, 17, 14, 30)} />
</div>

# Labels and format

Every word comes from `labels`, and `formatDate` writes the field's text. `actions` adds Today and Clear under the calendar.

<div className="w-72 max-w-full">
  <DatePicker
    locale={de}
    actions
    placeholder="Datum wählen"
    formatDate={date => date.toLocaleDateString('de-DE', { weekday: 'short', day: 'numeric', month: 'long', year: 'numeric' })}
    labels={{ date: 'Datum', close: 'Schließen', today: 'Heute', clear: 'Leeren', calendar: 'Kalender öffnen', invalid: 'Ungültiges Datum' }}
  />
</div>

# Invalid value

A `value` that reads as no date (a malformed string, an Invalid Date) shows the invalid state and the `labels.invalid` caption.

<div className="flex w-72 max-w-full flex-col gap-3">
  <div>
    <DatePicker defaultValue="2026-02-31-oops" />
  </div>
  <div>
    <DatePicker editable defaultValue="not a date" />
  </div>
</div>

# Read only

`readOnly` shows the value but never opens or takes typing. `value` may be a string.

<div className="flex w-72 max-w-full flex-col gap-2">
  <DatePicker readOnly defaultValue="2026-10-09" />
  <DatePicker readOnly editable defaultValue="2026-10-09" />
</div>

# Inside a dialog

The panel lifts above the dialog. Escape in the field first undoes a half-typed date (the dialog stays), then closes the dialog.

<Dialog>
  <Dialog.Trigger render={<Button variant="secondary">Edit contract</Button>} />
  <Dialog.Content>
    <Dialog.Header>
      <Dialog.Title>Edit contract</Dialog.Title>
      <Dialog.Description>Type the start date, or pick it from the calendar.</Dialog.Description>
    </Dialog.Header>
    <div className="flex flex-col gap-1">
      <Label htmlFor="dialog-start">Start date</Label>
      <DatePicker id="dialog-start" editable defaultValue="2026-10-09" />
    </div>
    <Dialog.Footer>
      <Dialog.Close render={<Button variant="ghost">Cancel</Button>} />
      <Button>Save</Button>
    </Dialog.Footer>
  </Dialog.Content>
</Dialog>
