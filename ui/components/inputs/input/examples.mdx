import Input, { InputGroup, Label } from './index'
import Textarea from '../textarea'
import { Mail, Search } from 'lucide-react'

# Basic

A single input with a leading icon.

<div className="w-72 max-w-full">
  <Input placeholder="Search teamspaces..." icon={Search} />
</div>

# Variants

`borderless` is invisible until hovered or focused; `bordered` is framed at rest.

<div className="flex flex-col gap-3 w-72 max-w-full">
  <Input placeholder="Borderless" />
  <Input variant="bordered" placeholder="Bordered" />
</div>

# With a Label

`Label` pairs with the input through `htmlFor` and `id`.

<div className="flex flex-col gap-1 w-72 max-w-full">
  <Label htmlFor="input-email">Email</Label>
  <Input id="input-email" placeholder="you@example.com" icon={Mail} />
</div>

# Invalid

`invalid` turns the field destructive.

<div className="flex flex-col gap-1 w-72 max-w-full">
  <Label htmlFor="input-invalid" invalid>
    Email
  </Label>
  <Input id="input-invalid" placeholder="you@example.com" icon={Mail} defaultValue="bad@" invalid />
</div>

# Group

`InputGroup` stacks labelled fields; the hover highlight follows the pointer from field to field.

<InputGroup>
  <InputGroup.Field index={0} label="Name" placeholder="Your name" />
  <InputGroup.Field index={1} label="Email" placeholder="you@example.com" icon={Mail} />
</InputGroup>

# Group with a Textarea

A field's children replace its Input: a `Textarea` joins the group's hover and takes the field's error.

<InputGroup>
  <InputGroup.Field index={0} label="Subject" placeholder="What's it about?" />
  <InputGroup.Field index={1} label="Message">
    <Textarea placeholder="Write your message" />
  </InputGroup.Field>
</InputGroup>

# Group Error

An `error` message renders below the field and marks it invalid.

<InputGroup>
  <InputGroup.Field
    index={0}
    label="Email"
    placeholder="you@example.com"
    icon={Mail}
    defaultValue="bad@"
    error="Please enter a valid email address."
  />
</InputGroup>
